import unittest
from datetime import date

from household import due_dates, split_bill


class SplitBill(unittest.TestCase):
    def test_even_split(self):
        self.assertEqual(split_bill(1200, 4), [300, 300, 300, 300])

    def test_remainder_goes_to_the_first(self):
        self.assertEqual(split_bill(1001, 2), [501, 500])

    def test_three_ways(self):
        self.assertEqual(split_bill(1000, 3), [333, 333, 333])

    def test_nobody(self):
        with self.assertRaises(ValueError):
            split_bill(500, 0)


class DueDates(unittest.TestCase):
    def test_same_day_each_month(self):
        self.assertEqual(due_dates(date(2026, 3, 15), 3), [date(2026, 3, 15), date(2026, 4, 15), date(2026, 5, 15)])

    def test_short_month(self):
        self.assertEqual(due_dates(date(2026, 1, 31), 2), [date(2026, 1, 31), date(2026, 2, 28)])

    def test_across_the_new_year(self):
        self.assertEqual(due_dates(date(2026, 11, 30), 3), [date(2026, 11, 30), date(2026, 12, 30), date(2027, 1, 30)])


if __name__ == "__main__":
    unittest.main()
